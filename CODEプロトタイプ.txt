<script>
var cv=document.getElementById('c'),g=cv.getContext('2d');
g.imageSmoothingEnabled=false;

var KW=48,TOP=200,BTOP=186,NW=15,G0=0.34;
var NOTE=[0,2,4,5,7,9,11];
var RB=['#e23b3b','#f08a2c','#f2d13c','#4caf50','#3d7fe0','#4b45b8','#8e44c4'];

var BLK=[];
for(var i=0;i<NW-1;i++){var m=i%7;
 if(m==0||m==1||m==3||m==4||m==5)BLK.push({x:i*KW+34,w:28,sm:NOTE[m]+1+12*((i/7)|0)});}

var STAGES=[
 {n:'HOT CROSS BUNS',m:[4,3,2,4,3,2,2,2,2,2,3,3,3,3,4,3,2]},
 {n:'MARY HAD A LITTLE LAMB',m:[4,3,2,3,4,4,4,3,3,3,4,6,6]},
 {n:'ODE TO JOY',m:[4,4,5,6,6,5,4,3,2,2,3,4,4,3,3]},
 {n:'TWINKLE TWINKLE LITTLE STAR',m:[2,2,6,6,7,7,6,5,5,4,4,3,3,2]}
];

var ac=null;
function audioInit(){
 if(!ac){var A=window.AudioContext||window.webkitAudioContext;if(A)ac=new A();}
 if(ac&&ac.state=='suspended')ac.resume();
}
function tone(f,d,v,tp){
 if(!ac)return;
 var t=ac.currentTime,o=ac.createOscillator(),gn=ac.createGain();
 o.type=tp||'triangle';o.frequency.value=f;
 gn.gain.setValueAtTime(0,t);
 gn.gain.linearRampToValueAtTime(v,t+0.012);
 gn.gain.exponentialRampToValueAtTime(0.0001,t+d);
 o.connect(gn);gn.connect(ac.destination);
 o.start(t);o.stop(t+d+0.05);
}
function smf(sm){return 261.626*Math.pow(2,sm/12);}
function wsm(w){return NOTE[w%7]+12*((w/7)|0);}

var S='title',si=0,mel=[],mi=0,ok=0,bad=0,score=0,combo=0;
var paint=[],parts=[],shake=0,flash=0,passed=false;
var p={wi:0,x:0,y:TOP,vx:0,vy:0,air:false,face:1,walk:0,wa:0,wb:0,wt:0,sq:0,cd:0};
var hl=false,hr=false,hj=false,charging=false,charge=0;

function startStage(n){
 si=n;mel=STAGES[n].m.slice();mi=0;ok=0;bad=0;score=0;combo=0;
 paint=[];for(var i=0;i<NW;i++)paint.push(null);
 parts=[];shake=0;flash=0;charging=false;charge=0;
 p.wi=mel[0];p.x=p.wi*KW+KW/2;p.y=TOP;p.vx=0;p.vy=0;p.air=false;
 p.face=1;p.walk=0;p.sq=0;p.cd=0;
 S='play';
}

function land(kind,idx){
 p.air=false;p.vy=0;p.vx=0;p.sq=1;
 if(kind=='b'){
  tone(smf(BLK[idx].sm),0.5,0.16,'square');
  bad++;combo=0;shake=6;
  return;
 }
 tone(smf(wsm(idx)),0.9,0.22);
 if(mi<mel.length&&idx==mel[mi]){
  paint[idx]=RB[idx%7];ok++;combo++;
  score+=100+combo*10;flash=10;
  for(var i=0;i<10;i++)parts.push({x:p.x,y:TOP-4,vx:(Math.random()-0.5)*3,vy:-Math.random()*3-1,c:RB[idx%7],l:30});
  mi++;
  if(mi>=mel.length){
   passed=(ok/(ok+bad))>=0.8;
   S='res';
  }
 }else{bad++;combo=0;shake=4;}
}

function surfAt(px,py,ny){
 for(var i=0;i<BLK.length;i++){
  var b=BLK[i];
  if(px>b.x&&px<b.x+b.w&&py<=BTOP&&ny>=BTOP)return{k:'b',i:i,y:BTOP};
 }
 if(py<=TOP&&ny>=TOP){
  var w=Math.floor(px/KW);if(w<0)w=0;if(w>NW-1)w=NW-1;
  return{k:'w',i:w,y:TOP};
 }
 return null;
}

function predict(dir,pw){
 var sx=p.x,sy=p.y,vx,vy;
 if(dir==0){vx=0;vy=-(4.5+2*pw);}else{vx=dir*(2.0+4.6*pw);vy=-(5.0+2.4*pw);}
 for(var t=0;t<400;t++){
  vy+=G0;var nx=sx+vx,ny=sy+vy;
  if(nx<10)nx=10;if(nx>NW*KW-10)nx=NW*KW-10;
  if(vy>0){var s=surfAt(nx,sy,ny);if(s)return{x:nx,y:s.y,k:s.k};}
  sx=nx;sy=ny;
 }
 return null;
}

function step(d){
 var t=p.wi+d;if(t<0||t>NW-1)return;
 p.wa=p.x;p.wb=t*KW+KW/2;p.wt=t;p.walk=0.001;p.face=d;
}

function launch(){
 var dir=hl?-1:(hr?1:0),pw=charge;
 audioInit();
 if(dir==0){p.vx=0;p.vy=-(4.5+2*pw);}
 else{p.vx=dir*(2.0+4.6*pw);p.vy=-(5.0+2.4*pw);p.face=dir;}
 p.air=true;
}

function upd(dt){
 if(shake>0)shake-=dt;
 if(flash>0)flash-=dt;
 for(var i=parts.length-1;i>=0;i--){
  var q=parts[i];q.x+=q.vx*dt;q.y+=q.vy*dt;q.vy+=0.12*dt;q.l-=dt;
  if(q.l<=0)parts.splice(i,1);
 }
 if(p.sq>0)p.sq-=0.12*dt;
 if(S!='play')return;

 if(charging){charge+=dt/36;if(charge>1)charge=1;}

 if(p.walk>0){
  p.walk+=dt*0.11;
  if(p.walk>=1){p.walk=0;p.x=p.wb;p.wi=p.wt;land('w',p.wt);}
  else p.x=p.wa+(p.wb-p.wa)*p.walk;
 }else if(!p.air){
  if(p.cd>0)p.cd-=dt;
  if(!charging&&p.cd<=0){
   if(hl){step(-1);p.cd=9;}
   else if(hr){step(1);p.cd=9;}
  }
 }

 if(p.air){
  p.vy+=G0*dt;
  var nx=p.x+p.vx*dt,ny=p.y+p.vy*dt;
  if(nx<10)nx=10;if(nx>NW*KW-10)nx=NW*KW-10;
  if(p.vy>0){
   var s=surfAt(nx,p.y,ny);
   if(s){
    p.x=nx;p.y=s.y;
    if(s.k=='w')p.wi=s.i;
    else p.wi=Math.max(0,Math.min(NW-1,Math.floor(nx/KW)));
    land(s.k,s.i);
    return;
   }
  }
  p.x=nx;p.y=ny;
 }
}

function rect(a,b,w,h,c){g.fillStyle=c;g.fillRect(a|0,b|0,w|0,h|0);}

function drawUni(px,py,f){
 g.save();g.translate(px,py);
 var sy=1+(p.sq>0?p.sq*0.18:0),sx=1-(p.sq>0?p.sq*0.14:0);
 g.scale(f*sx,sy);
 var W='#ffffff',D='#e8e4d8';
 rect(-10,-18,18,11,W);
 rect(4,-27,10,10,W);
 rect(12,-24,4,3,W);
 rect(10,-25,2,2,'#1a2133');
 rect(13,-34,3,8,'#f2d13c');
 rect(-8,-7,4,7,W);rect(2,-7,4,7,W);
 rect(-9,-2,5,2,D);rect(1,-2,5,2,D);
 rect(0,-30,5,4,RB[0]);rect(-4,-28,5,4,RB[1]);
 rect(-8,-26,5,4,RB[4]);rect(-11,-22,5,4,RB[6]);
 rect(-15,-19,5,4,RB[3]);
 g.restore();
}

function drw(){
 g.clearRect(0,0,480,270);
 rect(0,0,480,270,'#1a2133');
 var cam=p.x-240;if(cam<0)cam=0;if(cam>NW*KW-480)cam=NW*KW-480;
 var sh=shake>0?(Math.random()-0.5)*shake:0;
 g.save();g.translate(sh,0);

 g.strokeStyle='#222c44';g.lineWidth=7;
 g.beginPath();g.arc(240-cam*0.3,190,150,Math.PI,0);g.stroke();
 g.strokeStyle='#20293e';g.lineWidth=7;
 g.beginPath();g.arc(240-cam*0.3,190,132,Math.PI,0);g.stroke();

 rect(0,252,480,14,'#2a4a60');
 rect(0,252,480,3,'#3d6480');
 rect(0,266,480,4,'#111726');

 for(var i=0;i<NW;i++){
  var X=i*KW-cam;if(X>480||X<-KW)continue;
  var c=paint[i]||'#efece2';
  rect(X,TOP,KW,52,c);
  rect(X,TOP,KW,4,paint[i]?'#ffffff':'#ffffff');
  rect(X+KW-2,TOP,2,52,'#0d1220');
 }

 if(S=='play'&&mi<mel.length){
  var t=mel[mi],TX=t*KW-cam;
  var near=Math.abs(t-p.wi)<=1;
  g.save();
  g.globalAlpha=0.55+0.35*Math.sin(Date.now()/180);
  g.shadowBlur=18;g.shadowColor=near?'#3d7fe0':'#f2d13c';
  rect(TX+2,TOP,KW-4,52,near?'#3d7fe0':'#f2d13c');
  g.restore();
 }

 for(var i=0;i<BLK.length;i++){
  var b=BLK[i],X=b.x-cam;if(X>480||X<-40)continue;
  rect(X,BTOP,b.w,14,'#181d2b');
  rect(X,BTOP,b.w,3,'#333a4d');
 }

 if(charging&&!p.air&&p.walk<=0){
  var dir=hl?-1:(hr?1:0),pr=predict(dir,charge);
  if(pr){
   g.save();g.globalAlpha=0.8;
   rect(pr.x-cam-7,pr.y-6,14,3,'#8fc3ff');
   rect(pr.x-cam-2,pr.y-11,4,4,'#8fc3ff');
   g.restore();
  }
  rect(p.x-cam-14,p.y-48,28,5,'#0d1220');
  rect(p.x-cam-13,p.y-47,26*charge,3,'#f2d13c');
 }

 for(var i=0;i<parts.length;i++){
  var q=parts[i];rect(q.x-cam-2,q.y-2,4,4,q.c);
 }

 if(S=='play'||S=='res')drawUni(p.x-cam,p.y,p.face);
 g.restore();

 rect(0,0,480,34,'rgba(15,20,34,0.85)');
 g.fillStyle='#cfe0ff';g.font='9px monospace';g.textAlign='left';
 g.fillText(STAGES[si].n,10,14);
 rect(10,20,150,5,'#233258');
 rect(10,20,150*(mi/mel.length),5,'#3d7fe0');
 g.textAlign='right';
 g.fillText('SCORE '+score,470,14);
 var acc=(ok+bad)?Math.round(ok/(ok+bad)*100):100;
 g.fillStyle=acc>=80?'#77c97a':'#f2645f';
 g.fillText('ACC '+acc+'%  (PASS 80%)',470,28);

 if(S=='title'){
  rect(0,0,480,270,'rgba(15,20,34,0.9)');
  g.textAlign='center';g.fillStyle='#fff';g.font='26px monospace';
  g.fillText('PIANO JUMPER',240,110);
  g.font='10px monospace';g.fillStyle='#8fc3ff';
  g.fillText('JUMP. PLAY. MAKE MUSIC.',240,132);
  g.fillStyle='#cfe0ff';
  g.fillText('ARROWS / BUTTONS = WALK  (1 STEP = 1 NOTE)',240,166);
  g.fillText('HOLD SPACE OR JUMP TO CHARGE, RELEASE TO LEAP',240,182);
  g.fillText('NO DIRECTION = HOP IN PLACE (REPEATED NOTE)',240,198);
  g.fillText('TAP OR PRESS SPACE TO START',240,228);
 }
 if(S=='res'){
  rect(0,0,480,270,'rgba(15,20,34,0.9)');
  g.textAlign='center';g.font='20px monospace';
  g.fillStyle=passed?'#77c97a':'#f2645f';
  g.fillText(passed?'STAGE CLEAR':'TRY AGAIN',240,110);
  g.font='11px monospace';g.fillStyle='#cfe0ff';
  var a2=(ok+bad)?Math.round(ok/(ok+bad)*100):0;
  g.fillText('ACCURACY  '+a2+'%',240,142);
  g.fillText('SCORE  '+score,240,160);
  g.fillText(passed&&si<STAGES.length-1?'TAP TO GO NEXT STAGE':'TAP TO RETRY',240,196);
  if(passed&&si>=STAGES.length-1)g.fillText('ALL SONGS COMPLETE!',240,196);
 }
}

var last=0;
function loop(t){
 var dt=(t-last)/16.667;last=t;
 if(!dt||dt>4)dt=1;
 upd(dt);drw();
 requestAnimationFrame(loop);
}

function confirm2(){
 audioInit();
 if(S=='title'){startStage(0);return true;}
 if(S=='res'){
  if(passed&&si<STAGES.length-1)startStage(si+1);
  else if(passed)startStage(0);
  else startStage(si);
  return true;
 }
 return false;
}
function jdown(){
 if(confirm2())return;
 if(S=='play'&&!p.air&&p.walk<=0&&!charging){charging=true;charge=0;}
}
function jup(){
 if(charging){charging=false;if(S=='play'&&!p.air&&p.walk<=0)launch();}
}

addEventListener('keydown',function(e){
 if(e.repeat)return;
 var k=e.key;
 if(k=='ArrowLeft'){hl=true;e.preventDefault();}
 else if(k=='ArrowRight'){hr=true;e.preventDefault();}
 else if(k==' '||k=='ArrowUp'){hj=true;jdown();e.preventDefault();}
});
addEventListener('keyup',function(e){
 var k=e.key;
 if(k=='ArrowLeft')hl=false;
 else if(k=='ArrowRight')hr=false;
 else if(k==' '||k=='ArrowUp'){hj=false;jup();}
});

function bind(el,dn,up){
 el.addEventListener('touchstart',function(e){e.preventDefault();dn();},{passive:false});
 el.addEventListener('touchend',function(e){e.preventDefault();up();},{passive:false});
 el.addEventListener('touchcancel',function(e){e.preventDefault();up();},{passive:false});
 el.addEventListener('mousedown',function(e){e.preventDefault();dn();});
 addEventListener('mouseup',function(){up();});
}
bind(document.getElementById('bl'),function(){audioInit();hl=true;},function(){hl=false;});
bind(document.getElementById('br'),function(){audioInit();hr=true;},function(){hr=false;});
bind(document.getElementById('bj'),jdown,jup);
cv.addEventListener('touchstart',function(e){e.preventDefault();confirm2();},{passive:false});
cv.addEventListener('mousedown',function(){confirm2();});

requestAnimationFrame(loop);
</script></body></html>
